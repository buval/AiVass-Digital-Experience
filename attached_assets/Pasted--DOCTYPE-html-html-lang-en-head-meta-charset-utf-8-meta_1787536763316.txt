      verdict='Full but lopsided. <b>'+strong.short.toLowerCase()+'</b> is carrying more than its share, and the thin spheres will eventually send the bill.';
    else if (st.round>=70)
      verdict='Evenly thin. Nothing is broken — <b>everything is running low</b>. Start at the core; the rest lifts with it.';
    else
      verdict='Uneven and stretched. Do not fix eight things. <b>Fix '+weak[0].short.toLowerCase()+' first</b> and let it pull its neighbours up.';

    var ringVals=RINGS.map(function(r){return {r:r, v:state[r.key]};});
    var thin=ringVals.slice().sort(function(a,b){return a.v-b.v;})[0];
    var pattern;
    if (state.kin<=4 && state.ground<=4 && state.craft>=7)
      pattern="The migration pattern: kin and ground are thin, and craft is carrying both. Work can hold a great deal of weight. It cannot hug you. Build one more anchor before work becomes the only thing standing.";
    else if (state.core<=3)
      pattern="Your core is the thinnest part of the structure. Start there and nowhere else — the outer rings will not hold on a body running on empty.";
    else
      pattern="Thinnest layer: "+thin.r.short.toLowerCase()+". "+thin.r.low;
    document.getElementById("pattern").textContent=pattern;

    var box=document.getElementById("actions");
    box.textContent="";
    var head=document.createElement("p");
    head.className="label"; head.textContent="Do these three. Nothing else.";
    box.appendChild(head);
    weak.concat([{short:thin.r.short, key:thin.r.key, low:thin.r.low}]).forEach(function(w){

      d.innerHTML='<div class="tag">'+w.short+' · '+state[w.key]+'/10</div><p>'+w.low+'</p>';
      box.appendChild(d);
    });
